local function parseLuck(str)
    if not str then return 0 end
    str=tostring(str):gsub("%s+",""):upper()
    if str=="" or str=="0" then return 0 end
    local mults={K=1e3,M=1e6,B=1e9,T=1e12,Q=1e15}
    local num,suf=str:match("^([%d%.]+)([KMBTQ]+)$")
    if num then return math.floor((tonumber(num) or 0)*(mults[suf] or 1)) else return tonumber(str) or 0 end
end
local function formatCompact(n)
    if n>=1e12 then return string.format("%.2fT",n/1e12)
    elseif n>=1e9 then return string.format("%.2fB",n/1e9)
    elseif n>=1e6 then return string.format("%.2fM",n/1e6)
    elseif n>=1e3 then return string.format("%.1fK",n/1e3) else return tostring(n) end
end
local function getEggTier(m)
    local ok,t = pcall(function() return m:GetAttribute("EggTier") end)
    if not ok or t==nil or t=="" then return "Unknown" end
    return tostring(t)
end

local rarityOptions={"All"}
local raritySet={["All"]=true}
local selectedRaritySet={["All"]=true}
local rarityDropdown=nil
local function getSelectedRarityDisplay()
    if selectedRaritySet["All"] then return "All" end
    if not next(selectedRaritySet) then return "All" end
    local t={} for k,_ in pairs(selectedRaritySet) do table.insert(t,k) end
    return table.concat(t,", ")
end
local function isRarityAllowed(tier)
    if selectedRaritySet["All"] then return true end
    if not next(selectedRaritySet) then return true end
    return selectedRaritySet[tier]==true
end
local function ensureRarity(tier)
    tier=tostring(tier or "Unknown")
    if tier=="" then tier="Unknown" end
    if not raritySet[tier] then
        raritySet[tier]=true
        table.insert(rarityOptions, tier)
        if rarityDropdown then pcall(function() if rarityDropdown.Refresh then rarityDropdown:Refresh(rarityOptions) elseif rarityDropdown.SetOptions then rarityDropdown:SetOptions(rarityOptions) end end) end
        return true
    end
    return false
end

-- SAFE FIX
local function safeGetPivot(obj)
    local ok,cf = pcall(function() return obj:GetPivot() end)
    if ok then return cf end
    return nil
end
local function safeGetPrimaryPart(obj)
    if not obj:IsA("Model") then return nil end
    local ok,pp = pcall(function() return obj.PrimaryPart end)
    if ok then return pp end
    return nil
end
local function getEggPos(m)
    if not m then return nil end
    local cf = safeGetPivot(m)
    if cf then return cf.Position end
    if m:IsA("Model") then
        local pp = safeGetPrimaryPart(m)
        if pp then return pp.Position end
    end
    if m:IsA("BasePart") then return m.Position end
    local ok,part = pcall(function() return m:FindFirstChildWhichIsA("BasePart", true) end)
    if ok and part then return part.Position end
    for _,v in ipairs(m:GetDescendants()) do if v:IsA("BasePart") then return v.Position end end
    return nil
end

local function findByKeyword(keyword)
    keyword = keyword:lower()
    local er = workspace:FindFirstChild("EggRuntime")
    if er then
        for _,v in ipairs(er:GetChildren()) do if v.Name:lower():find(keyword,1,true) then return v end end
        for _,v in ipairs(er:GetDescendants()) do if v.Name:lower():find(keyword,1,true) then return v end end
    end
    for _,v in ipairs(workspace:GetChildren()) do if v.Name:lower():find(keyword,1,true) then return v end end
    for _,v in ipairs(workspace:GetDescendants()) do if v.Name:lower():find(keyword,1,true) and (v:IsA("Model") or v:IsA("Folder")) then return v end end
    return nil
end

local function tpToPos(p)
    local fish=getRidingFish()
    if fish then pcall(function() fish:PivotTo(CFrame.new(p+Vector3.new(0,3,0))) end) end
    local h=getHRP() if h then h.CFrame=CFrame.new(p+Vector3.new(0,3,0)) end
end
local function tweenToPos(p)
    local fish=getRidingFish()
    local movePart=nil
    if fish then
        if fish:IsA("Model") then movePart = safeGetPrimaryPart(fish) end
        if not movePart then local ok,part = pcall(function() return fish:FindFirstChildWhichIsA("BasePart", true) end) if ok then movePart=part end end
    end
    if not movePart then movePart=getHRP() end
    if not movePart then return end
    if currentTween then pcall(function() currentTween:Cancel() end) end
    local dur=math.clamp((movePart.Position-p).Magnitude/tweenSpeed,0.15,8)
    currentTween=TweenService:Create(movePart,TweenInfo.new(dur,Enum.EasingStyle.Linear),{CFrame=CFrame.new(p)})
    currentTween:Play()
    currentTween.Completed:Wait()
    if fish then pcall(function() fish:PivotTo(CFrame.new(p+Vector3.new(0,3,0))) end) end
    local h=getHRP() if h then h.CFrame=CFrame.new(p+Vector3.new(0,3,0)) end
end
local function unstuckChar()
    pcall(function()
        local ch=LocalPlayer.Character
        if not ch then return end
        for _,v in ipairs(ch:GetDescendants()) do if v:IsA("BasePart") then v.Anchored=false end end
        local hum=ch:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand=false hum.Sit=false hum:ChangeState(Enum.HumanoidStateType.GettingUp) task.wait(0.05) hum:ChangeState(Enum.HumanoidStateType.Running) end
    end)
end

local function getEggArea(m)
    if not m then return "Unknown" end
    local cur=m
    for i=1,15 do if not cur then break end local nm=cur.Name:lower() if nm:find("kingkalolo") or nm:find("kalolo") then return "KingKaloloBoss" end if nm:find("kraken") then return "KrakenBoss" end if nm=="wildeggs" then return "Wild" end cur=cur.Parent end
    return "Wild"
end
local function isAreaAllowed(area) if selectedArea=="All" then return true end return area==selectedArea end
local function getAllEggs()
    local list,seen={},{} 
    local function add(m) if m and not seen[m] then local ok = pcall(function() return m:GetAttribute("EggLuck") end) if ok and m:GetAttribute("EggLuck")~=nil then seen[m]=true table.insert(list,m) end end end
    for _,m in ipairs(workspace:GetDescendants()) do if m:IsA("Model") then add(m) end end
    return list
end
local function getFilteredEggs()
    local all=getAllEggs() local out={}
    for _,m in ipairs(all) do local luck=tonumber(m:GetAttribute("EggLuck")) or 0 local tier=getEggTier(m) local area=getEggArea(m) ensureRarity(tier) if luck>=minLuckFilter and isRarityAllowed(tier) and isAreaAllowed(area) then table.insert(out,m) end end
    return out
end

local function doSteal(model)
    if not model or not model.Parent then return end
    local eggPos=getEggPos(model) if not eggPos then return end
    local stayCF=nil
    local onFish=getRidingFish()~=nil
    local savedSilent=useSilentMethod
    if onFish then useSilentMethod=false end
    if useSilentMethod then saveCam() stayCF=getHRP() and getHRP().CFrame or CFrame.new(BASE_POS) cloneModel=createClone(stayCF) lockCamToClone(cloneModel) setInvisible(true) else stayCF=getHRP() and getHRP().CFrame or nil end
    tpToPos(eggPos) task.wait(0.25) spamInteract(model) task.wait(0.15) tweenToPos(BASE_POS) task.wait(0.2) fireStoreLoop() unstuckChar()
    if useSilentMethod and stayCF then tweenToPos(stayCF.Position) task.wait(0.15) setInvisible(false) restoreCam() if cloneModel then pcall(function() cloneModel:Destroy() end) cloneModel=nil end
    elseif stayCF then tweenToPos(stayCF.Position) task.wait(0.15) end
    if onFish and stayCF then local fish=getRidingFish() if fish then pcall(function() fish:PivotTo(stayCF) end) end end
    useSilentMethod=savedSilent unstuckChar()
end

local function refreshEsp()
    if not isEspEggEnabled then clearAllEsp() return end
    for _,m in ipairs(getAllEggs()) do
        local luck=tonumber(m:GetAttribute("EggLuck")) or 0 local tier=getEggTier(m) local area=getEggArea(m) ensureRarity(tier)
        if luck>=minLuckFilter and isRarityAllowed(tier) and isAreaAllowed(area) then local kg=tonumber(m:GetAttribute("EggKg")) or 0 createEggEsp(m,luck,kg) else removeEggEsp(m) end
    end
end

local TokenTab=Window:Tab({Title="Token", Icon="T"})
local TokenAcc=TokenTab:Accordion({Title="Auto Collect Token", Desc="Farm ActiveTokens", Open=true})
TokenAcc:Toggle({Title="Auto Token",Desc="Auto teleport ke token (speed slider)",Flag="AutoToken",Default=false,Callback=function(s) isCollectingToken=s if s then Window.Notify("Auto Token","ON",CYAN,1.5) task.spawn(function() while isCollectingToken do pcall(function() local folder=workspace:FindFirstChild("ActiveTokens") if not folder then task.wait(1) return end local all=getAllTokens() if #all==0 then task.wait(1) return end local hrp=getHRP() table.sort(all, function(a,b) local pa=getTokenPos(a) local pb=getTokenPos(b) if not pa or not pb or not hrp then return false end return (hrp.Position-pa).Magnitude < (hrp.Position-pb).Magnitude end) for _,tok in ipairs(all) do if not isCollectingToken then break end if tok.Parent then doCollectToken(tok) task.wait(0.22) end end end) task.wait(0.3) end end) else Window.Notify("Auto Token","OFF",Color3.fromRGB(248,113,113),1) end end})
TokenAcc:Button({Title="Collect Nearest Token Once",Desc="Teleport ke token terdekat 1x",Callback=function() local all=getAllTokens() if #all==0 then Window.Notify("Token","Ga ada token",Color3.fromRGB(248,113,113),1) return end local hrp=getHRP() local best,closest=nil,math.huge for _,m in ipairs(all) do local p=getTokenPos(m) if p and hrp then local d=(hrp.Position-p).Magnitude if d<closest then closest=d best=m end end end if best then doCollectToken(best) end end})
